All hooks must be used inside a `<NexusProvider>`.

## useContracts

Fetches active contracts with TanStack Query.

```tsx
import { useContracts } from '@nexus-framework/react';

const { data, isLoading, error } = useContracts<IouPayload>({
  templateId: 'my-package:Iou:Iou',
  parties: ['Alice::abc'],      // optional
  filter: { currency: 'USD' }, // optional
  fetchAll: false,              // default — first page only
  staleTime: 5_000,             // override query staleTime
});

// data?.contracts: ActiveContract<IouPayload>[]
// data?.nextPageToken?: string
```

### useContractsSuspense

Suspense variant — throws a promise while loading. Wrap in `<Suspense>`.

```tsx
const { data } = useContractsSuspense<IouPayload>({
  templateId: 'my-package:Iou:Iou',
});
```

## useCreateContract

Mutation hook for creating a new Daml contract.
Automatically invalidates the matching template cache on success.

```tsx
import { useCreateContract } from '@nexus-framework/react';

const { mutate, isPending } = useCreateContract({
  onSuccess: (result) => console.log('txn:', result.transactionId),
  onError: (err) => console.error(err),
});

mutate({
  templateId: 'my-package:Iou:Iou',
  createArguments: { owner: 'Alice::abc', amount: '100', currency: 'USD' },
  actAs: ['Alice::abc'],
});
```

## useExerciseChoice

Mutation hook for exercising a choice on an existing contract.

```tsx
import { useExerciseChoice } from '@nexus-framework/react';

const { mutate } = useExerciseChoice({
  invalidateTemplates: ['my-package:Iou:Iou'],
});

mutate({
  templateId: 'my-package:Iou:Iou',
  contractId: 'abc123',
  choice: 'Transfer',
  choiceArgument: { newOwner: 'Bob::xyz' },
  actAs: ['Alice::abc'],
});
```

## useLedgerMutation

Generic hook for custom multi-command batches.

```tsx
import { useLedgerMutation } from '@nexus-framework/react';

const { mutate } = useLedgerMutation({
  mutationFn: async (nexus, vars: { amount: string }) =>
    nexus.ledger.commands.createContract(
      'my-package:Iou:Iou',
      { amount: vars.amount, ... },
      ['Alice::abc'],
    ),
  invalidateTemplates: ['my-package:Iou:Iou'],
});
```

## SignUIT example: MarginCall workflow

This pattern shows how PrimeBank issues a `MarginCall` and VantageCapital responds with an `AllocationRecord` — all through Nexus Framework hooks.

```tsx title="MarginCallList.tsx"
"use client";

import { nexus } from '@/lib/nexus-client';

interface MarginCallPayload {
  counterparty: string;   // PrimeBank party
  institution: string;    // VantageCapital party
  operator: string;       // SignUIT party
  callAmount: string;     // Daml Numeric
  currency: string;
  dueDate: string;        // ISO 8601
  status: string;         // "Pending" | "Fulfilled"
}

interface AllocationRecordPayload {
  institution: string;
  operator: string;
  marginCallCid: string;  // contractId of the resolved MarginCall
  allocatedAssets: { assetType: string; quantity: string }[];
  approvedAt: string;
}

export function MarginCallManager() {
  const institutionParty = process.env.NEXT_PUBLIC_INSTITUTION_PARTY_ID!;

  // Query pending margin calls visible to the institution
  const { data: callsData } = nexus.useContracts<MarginCallPayload>({
    templateId: 'nexus-example-0.0.1:CollateralRouter:MarginCall',
    parties: [institutionParty],
  });

  // Mutation: approve allocation — institution is sole decision-maker
  const { mutate: approveAllocation, isPending } = nexus.useCreateContract({
    onSuccess: () => console.log('AllocationRecord created'),
  });

  const pendingCalls = callsData?.contracts.filter(
    (c) => c.payload.status === 'Pending'
  ) ?? [];

  function handleApprove(callContractId: string, callAmount: string) {
    approveAllocation({
      templateId: 'nexus-example-0.0.1:CollateralRouter:AllocationRecord',
      createArguments: {
        institution: institutionParty,
        marginCallCid: callContractId,
        allocatedAssets: [{ assetType: 'UST_10Y', quantity: callAmount }],
        approvedAt: new Date().toISOString(),
      },
      actAs: [institutionParty], // institution signs alone — no operator co-sig
    });
  }

  return (
    <ul>
      {pendingCalls.map((c) => (
        <li key={c.contractId}>
          <strong>{c.payload.callAmount} {c.payload.currency}</strong>
          {' '}due {c.payload.dueDate}
          <button
            onClick={() => handleApprove(c.contractId, c.payload.callAmount)}
            disabled={isPending}
          >
            Approve Allocation
          </button>
        </li>
      ))}
    </ul>
  );
}
```

<Callout type="info">
  `actAs: [institutionParty]` — the institution signs the `AllocationRecord` alone. This enforces the ownership model: operator observes, institution decides.
</Callout>
```
