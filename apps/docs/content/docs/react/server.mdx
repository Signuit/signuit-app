Server-side utilities live in `@nexus-framework/react/server`. **Never import this in Client Components.**

```ts
import {
  createServerNexusClient,
  getLedgerData,
  withLedgerAction,
  createTanStackLedgerContext,
} from '@nexus-framework/react/server';
```

## createServerNexusClient

Reads the session cookie from a `Request` and returns a `NexusClient` scoped to that user's party. Use in Server Components, `createServerFn`, and Server Actions.

<Tabs items={["TanStack Start", "Next.js"]}>
  <Tab value="TanStack Start">
  ```ts title="src/routes/dashboard/index.tsx"
  import { createServerFn } from '@tanstack/react-start';
  import { createServerNexusClient } from '@nexus-framework/react/server';

  const getDashboardData = createServerFn({ method: 'GET' }).handler(
    async ({ request }) => {
      const client = await createServerNexusClient({
        request,
        config: { ledgerApiUrl: process.env.CANTON_API_URL! },
      });

      const { contracts } = await client.ledger.contracts.fetchActiveContracts({
        templateId: 'nexus-example-0.0.1:CollateralRouter:CollateralHolding',
      });

      return { holdings: contracts };
    }
  );
  ```
  </Tab>
  <Tab value="Next.js">
  ```ts title="app/dashboard/page.tsx"
  import { cookies } from 'next/headers';
  import { createServerNexusClient } from '@nexus-framework/react/server';

  export default async function DashboardPage() {
    const client = await createServerNexusClient({
      request: new Request('http://localhost', {
        headers: { cookie: (await cookies()).toString() },
      }),
      config: { ledgerApiUrl: process.env.CANTON_API_URL! },
    });

    const { contracts } = await client.ledger.contracts.fetchActiveContracts({
      templateId: 'nexus-example-0.0.1:CollateralRouter:CollateralHolding',
    });

    return <pre>{JSON.stringify(contracts, null, 2)}</pre>;
  }
  ```
  </Tab>
</Tabs>

## SSR Prefetch + HydrationBoundary

Pre-populate the TanStack Query cache on the server so Client Components render with zero loading state.

<Tabs items={["TanStack Start", "Next.js"]}>
  <Tab value="TanStack Start">

  Use `createServerFn` to load data, then pass it as loader data to the route:

  ```tsx title="src/routes/dashboard/index.tsx"
  import { createFileRoute } from '@tanstack/react-router';
  import { createServerFn } from '@tanstack/react-start';
  import { dehydrate, HydrationBoundary, QueryClient } from '@tanstack/react-query';
  import { contractQueryOptions } from '@nexus-framework/react';
  import { createServerNexusClient } from '@nexus-framework/react/server';
  import { CollateralList } from './-components/CollateralList';

  const serverLoader = createServerFn({ method: 'GET' }).handler(
    async ({ request }) => {
      const client = await createServerNexusClient({
        request,
        config: { ledgerApiUrl: process.env.CANTON_API_URL! },
      });

      const queryClient = new QueryClient();
      await queryClient.prefetchQuery(
        contractQueryOptions({
          client,
          templateId: 'nexus-example-0.0.1:CollateralRouter:CollateralHolding',
        })
      );

      return { dehydratedState: dehydrate(queryClient) };
    }
  );

  export const Route = createFileRoute('/dashboard/')({
    loader: () => serverLoader(),
    component: function DashboardPage() {
      const { dehydratedState } = Route.useLoaderData();
      return (
        <HydrationBoundary state={dehydratedState}>
          <CollateralList /> {/* nexus.useContracts() hits cache — no loading state */}
        </HydrationBoundary>
      );
    },
  });
  ```

  </Tab>
  <Tab value="Next.js">
  ```tsx title="app/dashboard/page.tsx"
  import { contractQueryOptions } from '@nexus-framework/react';
  import { createServerNexusClient } from '@nexus-framework/react/server';
  import { dehydrate, HydrationBoundary, QueryClient } from '@tanstack/react-query';
  import { cookies } from 'next/headers';
  import { CollateralList } from '@/components/CollateralList';

  export default async function DashboardPage() {
    const client = await createServerNexusClient({
      request: new Request('http://localhost', {
        headers: { cookie: (await cookies()).toString() },
      }),
      config: { ledgerApiUrl: process.env.CANTON_API_URL! },
    });

    const queryClient = new QueryClient();
    await queryClient.prefetchQuery(
      contractQueryOptions({
        client,
        templateId: 'nexus-example-0.0.1:CollateralRouter:CollateralHolding',
      })
    );

    return (
      <HydrationBoundary state={dehydrate(queryClient)}>
        <CollateralList /> {/* renders instantly — no loading state */}
      </HydrationBoundary>
    );
  }
  ```
  </Tab>
</Tabs>

## withLedgerAction

Wraps a ledger operation with typed error handling. Returns a serializable `LedgerActionResult<T>` — safe to return from Server Actions.

<Tabs items={["TanStack Start", "Next.js"]}>
  <Tab value="TanStack Start">
  ```ts title="src/routes/api/approve-routing.ts"
  import { createServerFn } from '@tanstack/react-start';
  import { withLedgerAction, createServerNexusClient } from '@nexus-framework/react/server';

  export const approveRoutingSuggestion = createServerFn({ method: 'POST' })
    .validator((data: { contractId: string; institutionPartyId: string }) => data)
    .handler(async ({ request, data }) => {
      const client = await createServerNexusClient({
        request,
        config: { ledgerApiUrl: process.env.CANTON_API_URL! },
      });

      return withLedgerAction(client, (c) =>
        c.ledger.commands.exerciseChoice(
          'nexus-example-0.0.1:CollateralRouter:RoutingSuggestion',
          data.contractId,
          'ApproveSuggestion',
          {},
          [data.institutionPartyId],
        )
      );
    });
  ```

  Client usage:

  ```tsx title="src/routes/dashboard/-components/ApproveButton.tsx"
  import { approveRoutingSuggestion } from '@/routes/api/approve-routing';

  export function ApproveButton({ contractId, institutionPartyId }: Props) {
    const [isPending, startTransition] = useTransition();

    return (
      <button
        disabled={isPending}
        onClick={() =>
          startTransition(async () => {
            const result = await approveRoutingSuggestion({
              data: { contractId, institutionPartyId },
            });
            if (!result.success) console.error(result.error);
          })
        }
      >
        {isPending ? 'Approving…' : 'Approve'}
      </button>
    );
  }
  ```
  </Tab>
  <Tab value="Next.js">
  ```ts title="app/dashboard/actions.ts"
  'use server';

  import { withLedgerAction, createServerNexusClient } from '@nexus-framework/react/server';
  import { cookies } from 'next/headers';
  import { revalidatePath } from 'next/cache';

  export async function approveRoutingSuggestionAction(contractId: string, partyId: string) {
    const client = await createServerNexusClient({
      request: new Request('http://localhost', {
        headers: { cookie: (await cookies()).toString() },
      }),
      config: { ledgerApiUrl: process.env.CANTON_API_URL! },
    });

    const result = await withLedgerAction(client, (c) =>
      c.ledger.commands.exerciseChoice(
        'nexus-example-0.0.1:CollateralRouter:RoutingSuggestion',
        contractId,
        'ApproveSuggestion',
        {},
        [partyId],
      )
    );

    if (result.success) revalidatePath('/dashboard');
    return result;
  }
  ```
  </Tab>
</Tabs>

### LedgerActionResult

```ts
type LedgerActionResult<T> =
  | { success: true; data: T }
  | { success: false; error: string; code: 'LEDGER_ERROR' | 'UNKNOWN_ERROR' };
```

## createTanStackLedgerContext

Creates a TanStack Start-native ledger proxy handler. Mounts at `/api/ledger` and forwards authenticated requests to Canton — reading the session cookie automatically.

```ts title="src/routes/api/ledger.$.ts"
import { createServerFileRoute } from '@tanstack/react-start/server';
import { createTanStackLedgerContext } from '@nexus-framework/react/server';
import { nexus } from '@/lib/nexus-server';

const ledgerHandler = createTanStackLedgerContext(nexus);

export const ServerRoute = createServerFileRoute('/api/ledger/$').methods({
  GET: ({ request }) => ledgerHandler(request),
  POST: ({ request }) => ledgerHandler(request),
});
```

This is the bridge between the browser client (`baseUrl: '/api/ledger'`) and the Canton JSON Ledger API. All requests are authenticated via the session cookie — the token never leaves the server.
